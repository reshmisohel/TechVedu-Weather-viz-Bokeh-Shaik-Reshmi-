import pandas as pd
import os
from bokeh.plotting import figure, output_file, save
from bokeh.models import ColumnDataSource, LinearColorMapper
from bokeh.transform import transform
from bokeh.palettes import Viridis256

os.makedirs('output', exist_ok=True)
print("Loading data...")
data = pd.read_csv('weather_100k_2026.csv')
print(f"Loaded {data.shape}")

# 1. Scatter
print("Creating scatter...")
df_plot = data.sample(n=15000, random_state=42) if len(data) > 15000 else data
source = ColumnDataSource(df_plot)
mapper = LinearColorMapper(palette=Viridis256, low=df_plot['wind_speed'].min(), high=df_plot['wind_speed'].max())
p1 = figure(title="Temp vs Humidity - 2026", width=900, height=500, tools="pan,wheel_zoom,box_zoom,reset,save,hover")
p1.scatter(x='temperature', y='humidity', source=source, size=5, alpha=0.6, color=transform('wind_speed', mapper))
output_file('output/scatter.html')
save(p1)

# 2. Bar
print("Creating bar...")
monthly = data.groupby('month')['temperature'].mean().reset_index()
monthly['month_str'] = monthly['month'].astype(str)
source2 = ColumnDataSource(monthly)
p2 = figure(x_range=monthly['month_str'].tolist(), title="Monthly Avg Temp - 2026", width=900, height=500, tools="pan,wheel_zoom,box_zoom,reset,save,hover")
p2.vbar(x='month_str', top='temperature', width=0.5, source=source2)
output_file('output/bar.html')
save(p2)

# 3. Heatmap
print("Creating heatmap...")
corr = data[['temperature','humidity','wind_speed','precipitation']].corr().stack().reset_index()
corr.columns = ['x','y','value']
source3 = ColumnDataSource(corr)
p3 = figure(title="Correlation Heatmap - 2026", x_range=list(corr['x'].unique()), y_range=list(corr['y'].unique()), width=600, height=500, tools="pan,hover,save,reset")
p3.rect(x='x', y='y', width=1, height=1, source=source3, fill_color=transform('value', LinearColorMapper(palette=Viridis256, low=-1, high=1)), line_color=None)
output_file('output/heatmap.html')
save(p3)

print("ALL DONE - 3 HTML files saved in output/")
